```python
#!/usr/bin/env python3
"""
Demonstration Generator using Isaac Sim + MoveIt2

이 스크립트는:
1. Isaac Sim에서 로봇과 물체를 spawn
2. 목표 pose를 무작위로 샘플링
3. MoveIt2로 motion planning
4. 궤적을 실행하고 기록
5. Isaac Lab format으로 저장
"""

import omni
import numpy as np
from omni.isaac.kit import SimulationApp
import rclpy
from moveit_msgs.srv import GetMotionPlan

class DemoGenerator:
    def __init__(self, robot_name="hdr20_dg5f"):
        self.robot_name = robot_name
        self.sim_app = SimulationApp({"headless": True})

        # ROS 2 초기화
        rclpy.init()
        self.node = rclpy.create_node('demo_generator')

        # MoveIt2 service client
        self.plan_client = self.node.create_client(
            GetMotionPlan,
            '/plan_kinematic_path'
        )

        # 데이터 저장 경로
        self.save_dir = f"/home/dyros/IsaacLab/demos/{robot_name}/raw"

    def generate_target_pose(self):
        """무작위 목표 pose 생성"""
        # 물체 위치 주변에 목표 생성
        x = np.random.uniform(0.3, 0.5)
        y = np.random.uniform(-0.2, 0.2)
        z = np.random.uniform(0.1, 0.3)
        return [x, y, z]

    def plan_trajectory(self, target_pose):
        """MoveIt2로 궤적 계획"""
        request = GetMotionPlan.Request()
        # TODO: target_pose를 request에 설정

        future = self.plan_client.call_async(request)
        rclpy.spin_until_future_complete(self.node, future)

        if future.result().motion_plan_response.error_code.val == 1:
            return future.result().motion_plan_response.trajectory
        else:
            return None

    def execute_and_record(self, trajectory):
        """궤적을 실행하고 state-action pairs 기록"""
        states = []
        actions = []

        # TODO: Isaac Sim에서 궤적 실행하면서 기록

        return states, actions

    def save_demo(self, states, actions, demo_id):
        """Demonstration 저장"""
        np.savez(
            f"{self.save_dir}/demo_{demo_id:04d}.npz",
            states=np.array(states),
            actions=np.array(actions)
        )

    def generate_demos(self, num_demos=1000):
        """주요 루프: num_demos개 생성"""
        success_count = 0

        for i in range(num_demos):
            print(f"Generating demo {i+1}/{num_demos}...")

            # 목표 샘플링
            target_pose = self.generate_target_pose()

            # Motion planning
            trajectory = self.plan_trajectory(target_pose)
            if trajectory is None:
                print(f"  Planning failed, skipping...")
                continue

            # 실행 및 기록
            states, actions = self.execute_and_record(trajectory)

            # 저장
            self.save_demo(states, actions, success_count)
            success_count += 1

            if success_count % 100 == 0:
                print(f"✓ {success_count} demos generated successfully")

        print(f"\n✅ Total: {success_count}/{num_demos} demos generated")

if __name__ == "__main__":
    generator = DemoGenerator(robot_name="hdr20_dg5f")
    generator.generate_demos(num_demos=1000)
```

```python
def generate_synthetic_trajectory(start, goal, duration=5.0, dt=0.01):
    """부드러운 궤적 생성"""
    t = np.arange(0, duration, dt)
    waypoints = np.linspace(start, goal, 10)
    cs = CubicSpline(np.linspace(0, duration, 10), waypoints)
    return cs(t)
```

```python
#!/usr/bin/env python3
"""
Isaac Sim demonstration을 Isaac Lab format으로 변환
"""

import numpy as np
import h5py
from pathlib import Path

def convert_demo(demo_path, output_path):
    """
    Isaac Lab format:
    {
        'observations': {
            'policy': [...],      # robot joint pos/vel
            'proprio': [...],     # fingertip poses, contact
        },
        'actions': [...],         # joint position commands
        'rewards': [...],
        'dones': [...]
    }
    """
    # Load raw demo
    data = np.load(demo_path)
    states = data['states']
    actions = data['actions']

    # Convert to Isaac Lab format
    num_steps = len(states)

    observations = {
        'policy': np.zeros((num_steps, 64)),  # TODO: 정확한 dim
        'proprio': np.zeros((num_steps, 128))
    }

    # TODO: states -> observations 변환

    # Save as HDF5
    with h5py.File(output_path, 'w') as f:
        obs_grp = f.create_group('observations')
        obs_grp.create_dataset('policy', data=observations['policy'])
        obs_grp.create_dataset('proprio', data=observations['proprio'])
        f.create_dataset('actions', data=actions)
        f.create_dataset('rewards', data=np.zeros(num_steps))
        f.create_dataset('dones', data=np.zeros(num_steps, dtype=bool))

    print(f"✓ Converted: {output_path}")

def convert_all_demos(robot_name="hdr20_dg5f"):
    raw_dir = Path(f"/home/dyros/IsaacLab/demos/{robot_name}/raw")
    processed_dir = Path(f"/home/dyros/IsaacLab/demos/{robot_name}/processed")

    demo_files = sorted(raw_dir.glob("demo_*.npz"))
    print(f"Found {len(demo_files)} demos to convert")

    for demo_file in demo_files:
        output_file = processed_dir / f"{demo_file.stem}.h5"
        convert_demo(demo_file, output_file)

    print(f"\n✅ Converted {len(demo_files)} demos")

if __name__ == "__main__":
    convert_all_demos("hdr20_dg5f")
    convert_all_demos("uh035_inspire")
```

```python
#!/usr/bin/env python3
"""Demonstration 품질 검증"""

import h5py
import numpy as np
from pathlib import Path
import matplotlib.pyplot as plt

def validate_demo(demo_path):
    """단일 demo 검증"""
    with h5py.File(demo_path, 'r') as f:
        actions = f['actions'][:]

        # 체크 1: Action bounds
        if np.any(np.abs(actions) > 10.0):
            return False, "Action out of bounds"

        # 체크 2: 부드러운 궤적 (acceleration check)
        diff = np.diff(actions, axis=0)
        if np.max(np.abs(diff)) > 1.0:
            return False, "Trajectory not smooth"

        # 체크 3: 길이 체크
        if len(actions) < 10 or len(actions) > 1000:
            return False, f"Invalid length: {len(actions)}"

    return True, "OK"

def validate_all_demos(robot_name="hdr20_dg5f"):
    processed_dir = Path(f"/home/dyros/IsaacLab/demos/{robot_name}/processed")
    demo_files = sorted(processed_dir.glob("*.h5"))

    valid_count = 0
    invalid_demos = []

    for demo_file in demo_files:
        is_valid, msg = validate_demo(demo_file)
        if is_valid:
            valid_count += 1
        else:
            invalid_demos.append((demo_file.name, msg))

    print(f"\n{'='*60}")
    print(f"Validation Results for {robot_name}")
    print(f"{'='*60}")
    print(f"Total demos: {len(demo_files)}")
    print(f"Valid demos: {valid_count}")
    print(f"Invalid demos: {len(invalid_demos)}")

    if invalid_demos:
        print(f"\nInvalid demos:")
        for name, reason in invalid_demos[:10]:  # 처음 10개만 표시
            print(f"  - {name}: {reason}")

    # 통계 시각화
    visualize_demo_statistics(processed_dir)

    return valid_count >= 800  # 최소 800개 필요

def visualize_demo_statistics(demo_dir):
    """Demo 통계 시각화"""
    demo_files = sorted(demo_dir.glob("*.h5"))
    lengths = []

    for demo_file in demo_files[:100]:  # 샘플 100개만
        with h5py.File(demo_file, 'r') as f:
            lengths.append(len(f['actions']))

    plt.figure(figsize=(10, 5))
    plt.hist(lengths, bins=30)
    plt.xlabel('Trajectory Length (steps)')
    plt.ylabel('Count')
    plt.title('Distribution of Demonstration Lengths')
    plt.savefig(f'{demo_dir}/demo_statistics.png')
    print(f"✓ Saved statistics plot: {demo_dir}/demo_statistics.png")

if __name__ == "__main__":
    valid_hdr = validate_all_demos("hdr20_dg5f")
    valid_uh035 = validate_all_demos("uh035_inspire")

    if valid_hdr and valid_uh035:
        print("\n✅ Week 1 완료! Demonstrations 준비됨")
    else:
        print("\n⚠️ Warning: 일부 demos가 유효하지 않습니다")
```

```python
"""Contact sensor 데이터 확인"""

import torch
from isaaclab.app import AppLauncher
app = AppLauncher().app

from isaaclab_tasks.manager_based.manipulation.dexsuite_hdr import DexsuiteEnvCfg

def test_contact_sensors():
    env_cfg = DexsuiteEnvCfg()
    env = env_cfg.create_env(num_envs=4)

    env.reset()

    # 10 steps 실행
    for i in range(10):
        actions = torch.randn(4, env.num_actions, device=env.device)
        obs, rewards, dones, info = env.step(actions)

        # Contact forces 확인
        if hasattr(env, 'contact_forces'):
            contact_forces = env.contact_forces
            print(f"Step {i}: Contact forces shape: {contact_forces.shape}")
            print(f"  Max force: {contact_forces.max().item():.4f}")
            print(f"  Mean force: {contact_forces.mean().item():.4f}")

    print("\n✅ Contact sensors are working!")

if __name__ == "__main__":
    test_contact_sensors()
```

```python
def contact_gated_tracking(
    env,
    threshold: float = 0.1
) -> torch.Tensor:
    """
    위치 추적 보상을 손가락-물체 접촉이 있을 때만 활성화

    Args:
        env: Environment instance
        threshold: 최소 contact force (N)

    Returns:
        Gated reward tensor [num_envs]
    """
    # 1. 기존 tracking reward 계산
    object_pos = env.object.data.root_pos_w - env.scene.env_origins
    goal_pos = env.command_manager.get_command("object_pose")[:, :3]
    distance = torch.norm(object_pos - goal_pos, dim=-1)
    tracking_reward = 1.0 - torch.tanh(distance / 0.1)

    # 2. Contact 감지
    contact_forces = env.contact_sensor.data.net_forces_w_history
    fingertip_forces = contact_forces[:, env.fingertip_body_ids, :, :]  # [num_envs, num_fingers, 3, history]

    # 각 손가락의 최대 force
    max_finger_forces = torch.max(
        torch.norm(fingertip_forces, dim=2),  # [num_envs, num_fingers, history]
        dim=-1
    )[0]  # [num_envs, num_fingers]

    # 최소 1개 손가락이 접촉 중인가?
    any_contact = torch.any(max_finger_forces > threshold, dim=-1)  # [num_envs]

    # 3. Gating 적용
    gated_reward = tracking_reward * any_contact.float()

    return gated_reward
```

```python
def grasp_duration(
    env,
    min_duration: int = 10,  # 최소 지속 스텝
    force_threshold: float = 0.1
) -> torch.Tensor:
    """
    지속적인 다중 손가락 접촉에 대한 보상

    Args:
        env: Environment instance
        min_duration: 최소 지속 timesteps
        force_threshold: 접촉으로 간주할 최소 force

    Returns:
        Reward tensor [num_envs]
    """
    # Contact history 사용 (최근 min_duration steps)
    contact_forces = env.contact_sensor.data.net_forces_w_history  # [num_envs, num_bodies, 3, history_length]

    # Fingertip forces만 추출
    fingertip_forces = contact_forces[:, env.fingertip_body_ids, :, -min_duration:]  # 최근 min_duration steps

    # 각 timestep에서 접촉 여부 [num_envs, num_fingers, min_duration]
    finger_norms = torch.norm(fingertip_forces, dim=2)  # [num_envs, num_fingers, min_duration]
    is_contact = finger_norms > force_threshold  # Boolean

    # 몇 개 손가락이 전체 duration 동안 접촉했는가?
    sustained_contact = torch.all(is_contact, dim=-1)  # [num_envs, num_fingers]
    num_sustained_fingers = sustained_contact.float().sum(dim=-1)  # [num_envs]

    # 보상: 지속적으로 접촉한 손가락 수에 비례
    reward = num_sustained_fingers / 5.0  # Normalize by total fingers (5)

    return reward
```

```python
def ground_contact_penalty(
    env,
    penalty_scale: float = 1.0
) -> torch.Tensor:
    """
    물체-바닥 접촉에 대한 벌점

    Args:
        env: Environment instance
        penalty_scale: 벌점 스케일

    Returns:
        Penalty tensor [num_envs] (negative values)
    """
    # 물체의 contact forces
    object_contact = env.contact_sensor.data.net_forces_w[:, env.object_body_id, :]  # [num_envs, 3]

    # Z 방향 (위쪽) force가 크면 바닥과 충돌
    upward_force = object_contact[:, 2]  # [num_envs]

    # 충돌 감지 (upward force > threshold)
    is_ground_contact = upward_force > 1.0  # 1N 이상

    # 벌점 (negative reward)
    penalty = -penalty_scale * is_ground_contact.float()

    return penalty
```

```python
#!/usr/bin/env python3
"""Behavior Cloning Policy Network"""

import torch
import torch.nn as nn

class BCPolicy(nn.Module):
    """
    Behavior Cloning을 위한 policy network

    Input: observations (joint pos/vel, object pose, goal pose)
    Output: actions (joint position commands)
    """
    def __init__(
        self,
        obs_dim: int,
        action_dim: int,
        hidden_dims: list = [256, 256, 128]
    ):
        super().__init__()

        layers = []
        prev_dim = obs_dim

        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(prev_dim, hidden_dim))
            layers.append(nn.ReLU())
            layers.append(nn.LayerNorm(hidden_dim))
            prev_dim = hidden_dim

        layers.append(nn.Linear(prev_dim, action_dim))
        layers.append(nn.Tanh())  # Action range [-1, 1]

        self.network = nn.Sequential(*layers)

    def forward(self, obs):
        return self.network(obs)
```

```python
#!/usr/bin/env python3
"""Demonstration dataset for BC training"""

import h5py
import torch
from torch.utils.data import Dataset, DataLoader
from pathlib import Path

class DemoDataset(Dataset):
    """
    Demonstration dataset

    각 demo는 (state, action) pairs의 sequence
    """
    def __init__(self, demo_dir: str, robot_name: str = "hdr20_dg5f"):
        self.demo_dir = Path(demo_dir) / robot_name / "processed"
        self.demo_files = sorted(self.demo_dir.glob("*.h5"))

        print(f"Loading {len(self.demo_files)} demos from {self.demo_dir}")

        # 모든 demos를 메모리에 로드 (작은 데이터셋이므로 가능)
        self.states = []
        self.actions = []

        for demo_file in self.demo_files:
            with h5py.File(demo_file, 'r') as f:
                obs = f['observations']['policy'][:]
                acts = f['actions'][:]

                self.states.append(torch.from_numpy(obs).float())
                self.actions.append(torch.from_numpy(acts).float())

        # Concatenate all demos
        self.states = torch.cat(self.states, dim=0)
        self.actions = torch.cat(self.actions, dim=0)

        print(f"Total samples: {len(self.states)}")
        print(f"State dim: {self.states.shape[1]}")
        print(f"Action dim: {self.actions.shape[1]}")

    def __len__(self):
        return len(self.states)

    def __getitem__(self, idx):
        return self.states[idx], self.actions[idx]

def create_dataloaders(
    demo_dir: str,
    robot_name: str,
    train_split: float = 0.9,
    batch_size: int = 256
):
    """Train/val dataloaders 생성"""
    dataset = DemoDataset(demo_dir, robot_name)

    # Train/val split
    train_size = int(len(dataset) * train_split)
    val_size = len(dataset) - train_size

    train_dataset, val_dataset = torch.utils.data.random_split(
        dataset, [train_size, val_size]
    )

    train_loader = DataLoader(
        train_dataset,
        batch_size=batch_size,
        shuffle=True,
        num_workers=4
    )

    val_loader = DataLoader(
        val_dataset,
        batch_size=batch_size,
        shuffle=False,
        num_workers=2
    )

    return train_loader, val_loader
```

```python
#!/usr/bin/env python3
"""Behavior Cloning Trainer"""

import torch
import torch.nn as nn
import torch.optim as optim
from pathlib import Path
from tqdm import tqdm
import matplotlib.pyplot as plt

from bc_model import BCPolicy
from demo_dataset import create_dataloaders

class BCTrainer:
    def __init__(
        self,
        robot_name: str = "hdr20_dg5f",
        obs_dim: int = 64,  # TODO: 정확한 dim
        action_dim: int = 26,  # HDR20-DG5F: 26 DOF
        learning_rate: float = 3e-4,
        device: str = "cuda"
    ):
        self.robot_name = robot_name
        self.device = torch.device(device)

        # Model
        self.policy = BCPolicy(obs_dim, action_dim).to(self.device)
        print(f"Model parameters: {sum(p.numel() for p in self.policy.parameters()):,}")

        # Optimizer
        self.optimizer = optim.Adam(self.policy.parameters(), lr=learning_rate)

        # Loss
        self.criterion = nn.MSELoss()

        # Dataloaders
        self.train_loader, self.val_loader = create_dataloaders(
            demo_dir="/home/dyros/IsaacLab/demos",
            robot_name=robot_name,
            batch_size=256
        )

        # Logging
        self.train_losses = []
        self.val_losses = []

    def train_epoch(self):
        """1 epoch 훈련"""
        self.policy.train()
        total_loss = 0.0

        for states, actions in tqdm(self.train_loader, desc="Training"):
            states = states.to(self.device)
            actions = actions.to(self.device)

            # Forward
            pred_actions = self.policy(states)
            loss = self.criterion(pred_actions, actions)

            # Backward
            self.optimizer.zero_grad()
            loss.backward()
            torch.nn.utils.clip_grad_norm_(self.policy.parameters(), 1.0)
            self.optimizer.step()

            total_loss += loss.item()

        return total_loss / len(self.train_loader)

    def validate(self):
        """Validation"""
        self.policy.eval()
        total_loss = 0.0

        with torch.no_grad():
            for states, actions in self.val_loader:
                states = states.to(self.device)
                actions = actions.to(self.device)

                pred_actions = self.policy(states)
                loss = self.criterion(pred_actions, actions)

                total_loss += loss.item()

        return total_loss / len(self.val_loader)

    def train(self, num_epochs: int = 50):
        """전체 훈련 루프"""
        print(f"\n{'='*60}")
        print(f"Training BC Policy for {self.robot_name}")
        print(f"{'='*60}")
        print(f"Epochs: {num_epochs}")
        print(f"Train batches: {len(self.train_loader)}")
        print(f"Val batches: {len(self.val_loader)}")
        print(f"{'='*60}\n")

        best_val_loss = float('inf')
        patience = 10
        patience_counter = 0

        for epoch in range(num_epochs):
            train_loss = self.train_epoch()
            val_loss = self.validate()

            self.train_losses.append(train_loss)
            self.val_losses.append(val_loss)

            print(f"Epoch {epoch+1}/{num_epochs}")
            print(f"  Train Loss: {train_loss:.6f}")
            print(f"  Val Loss:   {val_loss:.6f}")

            # Early stopping
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                patience_counter = 0
                self.save_checkpoint(epoch, val_loss, is_best=True)
            else:
                patience_counter += 1

            if patience_counter >= patience:
                print(f"\nEarly stopping at epoch {epoch+1}")
                break

            # Regular checkpoint
            if (epoch + 1) % 10 == 0:
                self.save_checkpoint(epoch, val_loss, is_best=False)

        # Plot training curve
        self.plot_training_curve()

        print(f"\n✅ Training completed!")
        print(f"Best validation loss: {best_val_loss:.6f}")

    def save_checkpoint(self, epoch, val_loss, is_best=False):
        """모델 저장"""
        save_dir = Path(f"/home/dyros/IsaacLab/results/week3_bc/{self.robot_name}")
        save_dir.mkdir(parents=True, exist_ok=True)

        checkpoint = {
            'epoch': epoch,
            'model_state_dict': self.policy.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'val_loss': val_loss,
            'train_losses': self.train_losses,
            'val_losses': self.val_losses
        }

        if is_best:
            save_path = save_dir / "bc_policy_best.pth"
            print(f"  → Saved best model: {save_path}")
        else:
            save_path = save_dir / f"bc_policy_epoch{epoch+1}.pth"

        torch.save(checkpoint, save_path)

    def plot_training_curve(self):
        """Training curve 시각화"""
        plt.figure(figsize=(10, 5))
        plt.plot(self.train_losses, label='Train Loss')
        plt.plot(self.val_losses, label='Val Loss')
        plt.xlabel('Epoch')
        plt.ylabel('MSE Loss')
        plt.title(f'BC Training Curve - {self.robot_name}')
        plt.legend()
        plt.grid(True)

        save_path = f"/home/dyros/IsaacLab/results/week3_bc/{self.robot_name}/training_curve.png"
        plt.savefig(save_path)
        print(f"✓ Saved training curve: {save_path}")

if __name__ == "__main__":
    # HDR20-DG5F
    trainer_hdr = BCTrainer(
        robot_name="hdr20_dg5f",
```

```python
#!/usr/bin/env python3
"""RL training with BC initialization"""

import torch
from rl_games.torch_runner import Runner
from pathlib import Path

def load_bc_checkpoint_to_rl_policy(bc_checkpoint_path, rl_agent):
    """
    BC checkpoint를 RL agent의 policy network에 로드

    Args:
        bc_checkpoint_path: BC policy checkpoint path
        rl_agent: RL-Games agent
    """
    print(f"Loading BC checkpoint from: {bc_checkpoint_path}")

    checkpoint = torch.load(bc_checkpoint_path)
    bc_state_dict = checkpoint['model_state_dict']

    # RL agent의 actor network에 BC weights 복사
    rl_actor_dict = rl_agent.model.a2c_network.actor.state_dict()

    # BC network의 weights를 RL actor에 매칭
    for bc_key, rl_key in zip(bc_state_dict.keys(), rl_actor_dict.keys()):
        if bc_state_dict[bc_key].shape == rl_actor_dict[rl_key].shape:
            rl_actor_dict[rl_key] = bc_state_dict[bc_key]
            print(f"  Loaded: {bc_key} -> {rl_key}")
        else:
            print(f"  Skipped: {bc_key} (shape mismatch)")

    rl_agent.model.a2c_network.actor.load_state_dict(rl_actor_dict)

    print("✓ BC initialization complete!")

def train_rl_with_bc(
    variant: str,  # "pure_ppo", "bc_ppo", "ppo_contact", "bc_ppo_contact"
    robot_name: str = "hdr20_dg5f",
    num_epochs: int = 5000
):
    """
    RL training with specified variant

    Args:
        variant: Training variant
        robot_name: Robot name
        num_epochs: Training epochs
    """
    print(f"\n{'='*70}")
    print(f"Training: {variant} - {robot_name}")
    print(f"{'='*70}\n")

    # Config path
    config_path = f"source/isaaclab_tasks/.../config/{robot_name}/agents/rl_games_ppo_cfg.yaml"

    # Modify config based on variant
    if "bc" in variant:
        bc_checkpoint = f"/home/dyros/IsaacLab/results/week3_bc/{robot_name}/bc_policy_best.pth"
    else:
        bc_checkpoint = None

    if "contact" in variant:
        # Contact-aware rewards 활성화
        enable_contact_rewards = True
    else:
        enable_contact_rewards = False

    # Create RL-Games runner
    runner = Runner()
    runner.load(config_path)

    # BC initialization (if needed)
    if bc_checkpoint:
        load_bc_checkpoint_to_rl_policy(bc_checkpoint, runner.agent)

    # Start training
    runner.reset()
    runner.run({
        'train': True,
        'play': False,
        'checkpoint': None,
        'sigma': None
    })

    print(f"\n✅ Training completed: {variant} - {robot_name}")

if __name__ == "__main__":
    # HDR20-DG5F
    variants = ["pure_ppo", "bc_ppo", "ppo_contact", "bc_ppo_contact"]

    for variant in variants:
        train_rl_with_bc(variant, robot_name="hdr20_dg5f", num_epochs=5000)
```

```python
#!/usr/bin/env python3
"""Policy evaluation script"""

import torch
import numpy as np
from pathlib import Path
import json
from isaaclab.app import AppLauncher
app = AppLauncher().app

from isaaclab_tasks.manager_based.manipulation.dexsuite_hdr import DexsuiteEnvCfg

def evaluate_policy(
    checkpoint_path: str,
    robot_name: str = "hdr20_dg5f",
    num_episodes: int = 100,
    render: bool = False
):
    """
    단일 policy 평가

    Returns:
        results: dict with metrics
    """
    # Load checkpoint
    checkpoint = torch.load(checkpoint_path)

    # TODO: Policy 로드 (RL-Games format)

    # Create environment
    env_cfg = DexsuiteEnvCfg()
    env = env_cfg.create_env(num_envs=100, render=render)

    # Evaluation metrics
    success_rates = []
    episode_lengths = []
    contact_forces = []

    print(f"Evaluating: {checkpoint_path}")
    print(f"Episodes: {num_episodes}")

    for ep in range(num_episodes):
        obs = env.reset()
        done = False
        step = 0
        ep_forces = []

        while not done and step < 500:
            # Get action from policy
            with torch.no_grad():
                actions = policy(obs)

            obs, rewards, dones, info = env.step(actions)

            # Collect contact forces
            if hasattr(env, 'contact_sensor'):
                forces = env.contact_sensor.data.net_forces_w
                ep_forces.append(forces.cpu().numpy())

            done = dones.any()
            step += 1

        # Record metrics
        success = info.get('success', torch.zeros(100, dtype=bool))
        success_rates.append(success.float().mean().item())
        episode_lengths.append(step)
        contact_forces.extend(ep_forces)

        if (ep + 1) % 10 == 0:
            print(f"  Episodes {ep+1}/{num_episodes}: Success = {np.mean(success_rates):.2%}")

    # Aggregate results
    results = {
        'success_rate': np.mean(success_rates),
        'success_std': np.std(success_rates),
        'avg_episode_length': np.mean(episode_lengths),
        'avg_contact_force': np.mean([f.mean() for f in contact_forces]),
        'max_contact_force': np.max([f.max() for f in contact_forces])
    }

    print(f"\n{'='*60}")
    print(f"Evaluation Results")
    print(f"{'='*60}")
    for key, value in results.items():
        print(f"{key:25s}: {value:.4f}")
    print(f"{'='*60}\n")

    env.close()
    return results

def evaluate_all_experiments():
    """모든 실험 평가"""
    experiments = [
        ("hdr20_dg5f", "pure_ppo"),
        ("hdr20_dg5f", "bc_ppo"),
        ("hdr20_dg5f", "ppo_contact"),
        ("hdr20_dg5f", "bc_ppo_contact"),
        ("uh035_inspire", "pure_ppo"),
        ("uh035_inspire", "bc_ppo"),
        ("uh035_inspire", "ppo_contact"),
        ("uh035_inspire", "bc_ppo_contact"),
    ]

    all_results = {}

    for robot, variant in experiments:
        checkpoint_path = f"results/week4_rl/{robot}/{variant}/checkpoint_final.pth"

        if not Path(checkpoint_path).exists():
            print(f"⚠️ Skipping {robot}/{variant}: checkpoint not found")
            continue

        results = evaluate_policy(checkpoint_path, robot, num_episodes=100)
        all_results[f"{robot}_{variant}"] = results

    # Save results
    output_path = "results/week5_eval/evaluation_results.json"
    with open(output_path, 'w') as f:
        json.dump(all_results, f, indent=2)

    print(f"✓ Saved results: {output_path}")
    return all_results

if __name__ == "__main__":
```
